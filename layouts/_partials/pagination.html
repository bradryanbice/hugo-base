{{- /* Accessible pagination. Written here rather than using Hugo's embedded pagination partial, because this needs a labelled landmark, aria-current on the current page, text on the previous and next links, and a window that does not grow without bound. Context: the page being rendered (its .Paginator has already been built). The current page is a span, not a link: a link to the page you are already on is noise for a screen reader. Every number has a visually hidden "Page" prefix, so "3" is announced as "Page 3" rather than as a bare digit. */ -}} {{- $paginator := .Paginator -}} {{- if gt $paginator.TotalPages 1 -}} {{- $current := $paginator.PageNumber -}} {{- $total := $paginator.TotalPages -}} {{- /* Show the first and last page, and a window of two either side of the current one. Gaps become an ellipsis. */ -}} {{- $window := 2 -}}
  • {{- with $paginator.Prev }} {{ i18n "previousPage" }} {{- else }} {{ i18n "previousPage" }} {{- end }}
    • {{- $lastShown := 0 -}} {{- range $paginator.Pagers }} {{- $number := .PageNumber -}} {{- $show := or (eq $number 1) (eq $number $total) (le (math.Abs (sub $number $current)) $window) -}} {{- if $show }} {{- if gt $number (add $lastShown 1) }}
  • …
    • {{- end }}
  • {{- if eq $number $current }} {{ i18n "pageNumber" }} {{ $number }} {{- else }} {{ i18n "pageNumber" }} {{ $number }} {{- end }}
    • {{- $lastShown = $number -}} {{- end }} {{- end }}
  • {{- with $paginator.Next }} {{ i18n "nextPage" }} {{- else }} {{ i18n "nextPage" }} {{- end }}

{{ i18n "pageXOfY" (dict "current" $current "total" $total) }}

{{- end -}}
